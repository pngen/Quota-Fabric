# Run: each example is a small program demonstrating a Quota Fabric concept.
function(qf_example NAME)
  add_executable(example_${NAME} ${NAME}.cpp)
  target_link_libraries(example_${NAME} PRIVATE quota_fabric_core)
  target_compile_features(example_${NAME} PRIVATE cxx_std_20)
endfunction()
